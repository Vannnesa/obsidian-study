```c
struct rect_tag {
  float left;
  float bottom;
  float top;
  float right;
};
typedef struct rect_tag rect_t;

float minimum(float f1, float f2) {
  // 比较 f1 和 f2
  if (f1 < f2) {
    // 如果 f1 小于 f2，那么 f1 就是答案
    return f1;
  }
  else {
    // 否则，f2 就是答案
    return f2;
  }
}

float maximum(float f1, float f2) {
  // 比较 f1 和 f2
  if (f1 > f2) {
    // 如果 f1 大于 f2，那么 f1 就是答案
    return f1;
  }
  else {
    // 否则，f2 就是答案
    return f2;
  }
}

// 求两个矩形 r1 和 r2 的交集：
rect_t intersection(rect_t r1, rect_t r2) {
  // 创建一个矩形（称为 ans），具有：
  rect_t ans;
  // left: r1 的 left 和 r2 的 left 的最大值
  ans.left = maximum(r1.left, r2.left);
  // bottom: r1 的 bottom 和 r2 的 bottom 的最大值
  ans.bottom = maximum(r1.bottom, r2.bottom);
  // right: r1 的 right 和 r2 的 right 的最小值
  ans.right = minimum(r1.right, r2.right);
  // top: r1 的 top 和 r2 的 top 的最小值
  ans.top = minimum(r1.top, r2.top);
  // 称为 ans 的矩形就是你的答案
  return ans;
}
```
